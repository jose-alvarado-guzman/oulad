# Dependencies of the ETL pipeline. In Colab, install these before running the
# package:  !pip install -r requirements.txt
#
# Upper bounds are set at the next major version because pyneoinstance 4 and
# neo4j 6 both carried breaking changes; loosen them deliberately, not by
# accident.
pyneoinstance>=4.0.3,<5
pandas>=3.0,<4
python-dotenv>=1.2,<2

# PyYAML is imported directly, to catch a parse error on config.yaml. neo4j
# arrives with pyneoinstance and is pinned only to keep the driver on the major
# version the Cypher was written against.
PyYAML>=6.0,<7
neo4j>=6.2,<7

# Not imported here, but pinned to work around an upstream bound that is too
# loose. pyneoinstance imports neo4j-viz, whose widget module evaluates
# traitlets.Instance[...] at import time, and that only became subscriptable in
# traitlets 5.10. neo4j-viz asks for traitlets>=5,<6, so an environment already
# holding 5.7 (Colab pins exactly that) keeps it and `import pyneoinstance`
# dies with "type 'Instance' is not subscriptable".
traitlets>=5.10,<6

# neo4j-rust-ext is a hard requirement of pyneoinstance rather than an optional
# speedup, so pip installs it whether or not it is named here.

# Aura Graph Analytics work has its own credentials, AGA_SECRETS in
# oulad/credentials.py, and will have its own dependencies. Keep those out of
# this file so an ETL-only environment stays minimal.
